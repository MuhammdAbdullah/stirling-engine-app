# Copy and paste ALL of this into your Pi SSH session:

sudo systemctl stop stirling-engine.service
sudo chown root:root /opt/stirling-engine-app/chrome-sandbox
sudo chmod 4755 /opt/stirling-engine-app/chrome-sandbox
sudo tee /etc/systemd/system/stirling-engine.service > /dev/null << 'EOF'
[Unit]
Description=Matrix Stirling Engine Application
After=graphical.target
Wants=graphical.target

[Service]
Type=simple
User=abdullah
Environment="DISPLAY=:0"
Environment="XAUTHORITY=/home/abdullah/.Xauthority"
ExecStartPre=/bin/sleep 15
ExecStart=/usr/local/bin/matrix-stirling-engine
Restart=on-failure
RestartSec=10

[Install]
WantedBy=graphical.target
EOF
sudo systemctl daemon-reload
sudo systemctl restart stirling-engine.service
sleep 3
sudo systemctl status stirling-engine.service
ps aux | grep -v grep | grep stirling-engine-monitor
